// ── Particle System ──────────────────────────────────────────
const canvas = document.getElementById('particleCanvas');
const ctx = canvas.getContext('2d');
let particles = [];
let isGold = false;

function resize() {
  canvas.width = window.innerWidth;
  canvas.height = document.getElementById('hero-section').offsetHeight;
  initParticles();
}

function createParticle() {
  const p = {
    x: Math.random() * canvas.width,
    y: Math.random() * canvas.height,
    speed: Math.random() / 5 + 0.1,
    opacity: 1,
    fadingOut: false,
    fadeStart: Date.now() + Math.random() * 600 + 100,
    reset() {
      this.x = Math.random() * canvas.width;
      this.y = Math.random() * canvas.height;
      this.speed = Math.random() / 5 + 0.1;
      this.opacity = 1;
      this.fadingOut = false;
      this.fadeStart = Date.now() + Math.random() * 600 + 100;
    },
    update() {
      this.y -= this.speed;
      if (this.y < 0) this.reset();
      if (!this.fadingOut && Date.now() > this.fadeStart) this.fadingOut = true;
      if (this.fadingOut) { this.opacity -= 0.008; if (this.opacity <= 0) this.reset(); }
    },
    draw(ctx) {
      if (isGold) {
        ctx.fillStyle = `rgba(220,180,20,${this.opacity * 0.85})`;
      } else {
        const r = 255 - (Math.random() * 255) / 2;
        ctx.fillStyle = `rgba(${r},220,255,${this.opacity})`;
      }
      ctx.fillRect(this.x, this.y, 0.5, Math.random() * 2 + 1);
    }
  };
  return p;
}

function initParticles() {
  const count = Math.floor((canvas.width * canvas.height) / 6000);
  particles = [];
  for (let i = 0; i < count; i++) particles.push(createParticle());
}

function animate() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  particles.forEach(p => { p.update(); p.draw(ctx); });
  requestAnimationFrame(animate);
}

resize();
window.addEventListener('resize', resize);
animate();
